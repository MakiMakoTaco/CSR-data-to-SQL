letterToNumber(letter) {
  return letter.toLowerCase().charCodeAt(0) - 97
}

  - create player map:

each side/DLC tier =>
  - create side map:
    name (sideName) => {
      id: super-side = 1, [letter]-Side => letterToNumber(letter) + 2, catstare = 1000, DLC = 1000 + DLC number,
      type: standard, catstare, dlc, other,

super side = 0 * letterTiers.length (7) + index of tier name (bronze = 0, silver = 1) + base tier id (1)
letter side = letterToNumber + 1 * letterTiers.length (7) + index of tier name (bronze = 0, silver = 1) + base tier id (1)
catstare = catstareTier.length - 1 + base tier id (100000)
dlc = dlc number - 1 * dlctiers.length (7) + index of tier name (bronze = 0, silver = 1) + base tier id (101000)
